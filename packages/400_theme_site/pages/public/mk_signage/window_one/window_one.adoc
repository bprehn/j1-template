<!-- Modal Controller Script -->
<script>
document.addEventListener('DOMContentLoaded', () => {
    const weatherModalEl = document.getElementById('weatherClosedModal');
    let weatherModal = null;
    
    // Elements for dynamic text
    const line1El = document.getElementById('weather-line1');
    const line2El = document.getElementById('weather-line2');
    const line3El = document.getElementById('weather-line3');

    // Listen to the node in Firebase
    const database = firebase.database();
    database.ref('systemStatus/weatherClosed').on('value', snapshot => {
        const data = snapshot.val();
        
        let isEnabled = false;
        let line1 = 'CLOSED';
        let line2 = 'Due to Weather Conditions';
        let line3 = 'Please check our website for updates.';

        if (data) {
            // Handle backwards compatibility with the old boolean format during migration
            if (typeof data === 'boolean' || typeof data === 'string') {
                isEnabled = (data === true || data === "true" || data === "Closed");
            } else {
                isEnabled = data.enabled === true || data.enabled === "true";
                line1 = data.line1 || line1;
                line2 = data.line2 || line2;
                line3 = data.line3 || line3;
            }
        }
        
        // Update DOM text
        line1El.textContent = line1;
        line2El.textContent = line2;
        line3El.textContent = line3;

        // Initialize modal only when needed
        if (!weatherModal) {
            weatherModal = new bootstrap.Modal(weatherModalEl);
        }
        
        if (isEnabled) {
            weatherModal.show();
        } else {
            weatherModal.hide();
        }
    });

    // TEST HELPER: Expose a global function to easily toggle this from the browser console
    window.toggleWeatherClosure = function(isClosed) {
        database.ref('systemStatus/weatherClosed').set({
            enabled: isClosed,
            line1: "TEST CLOSURE",
            line2: "This is a test from the console.",
            line3: "Check back later!"
        })
        .then(() => console.log('Weather closure set to:', isClosed))
        .catch(e => console.error('Error:', e));
    };
});
</script>
